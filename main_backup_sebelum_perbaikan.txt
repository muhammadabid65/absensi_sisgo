class SisgoShell extends StatefulWidget {
  const SisgoShell({super.key});

  @override
  State<SisgoShell> createState() => _SisgoShellState();
}

class _SisgoShellState extends State<SisgoShell> {
  int _selectedPage = 0;
  bool _isSignedIn = false;
  bool _isCheckedIn = false;
  Uint8List? _checkInPhoto;
  final _emailController = TextEditingController();
  final _passwordController = TextEditingController();
  final _loginFormKey = GlobalKey<FormState>();

  @override
  void dispose() {
    _emailController.dispose();
    _passwordController.dispose();
    super.dispose();
  }

  Future<void> _takeCheckInPhoto() async {
    try {
      final photo = await ImagePicker().pickImage(
        source: ImageSource.camera,
        imageQuality: 78,
        maxWidth: 1400,
      );
      if (photo == null || !mounted) return;
      setState(() => _checkInPhoto = null);
      final bytes = await photo.readAsBytes();
      if (mounted) setState(() => _checkInPhoto = bytes);
    } catch (_) {
      if (!mounted) return;
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Kamera tidak dapat dibuka. Coba lagi.')),
      );
    }
  }

  void _signIn() {
    if (!_loginFormKey.currentState!.validate()) return;
    setState(() => _isSignedIn = true);
  }

  void _checkIn() {
    if (_checkInPhoto == null) {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Ambil foto terlebih dahulu untuk check-in.')),
      );
      return;
    }
    setState(() => _isCheckedIn = true);
    ScaffoldMessenger.of(context).showSnackBar(
      const SnackBar(content: Text('Check-in berhasil dicatat.')),
    );
  }

  @override
  Widget build(BuildContext context) {
    if (!_isSignedIn) {
      return _LoginPage(
        formKey: _loginFormKey,
        emailController: _emailController,
        passwordController: _passwordController,
        onSubmit: _signIn,
      );
    }

    return LayoutBuilder(
      builder: (context, constraints) {
        final isWide = constraints.maxWidth >= 760;
        final pages = [
          _HomePage(
            isCheckedIn: _isCheckedIn,
            checkInPhoto: _checkInPhoto,
            onTakePhoto: _takeCheckInPhoto,
            onCheckIn: _checkIn,
            onApplyLeave: _showLeaveForm,
            onShowHistory: () => setState(() => _selectedPage = 1),
          ),
          const _HistoryPage(),
          const _ProfilePage(),
        ];

        return Scaffold(
          body: SafeArea(
            child: Row(
              children: [
                if (isWide)
                  _SideNavigation(
                    selectedPage: _selectedPage,
                    onSelect: (index) => setState(() => _selectedPage = index),
                  ),
                Expanded(
                  child: Align(
                    alignment: Alignment.topCenter,
                    child: ConstrainedBox(
                      constraints: const BoxConstraints(maxWidth: 1080),
                      child: pages[_selectedPage],
                    ),
                  ),
                ),
              ],
            ),
          ),
          bottomNavigationBar: isWide
              ? null
              : NavigationBar(
                  selectedIndex: _selectedPage,
                  onDestinationSelected: (index) =>
                      setState(() => _selectedPage = index),
                  destinations: const [
                    NavigationDestination(
                      icon: Icon(Icons.home_outlined),
                      selectedIcon: Icon(Icons.home_rounded),
                      label: 'Beranda',
                    ),
                    NavigationDestination(
                      icon: Icon(Icons.event_note_outlined),
                      selectedIcon: Icon(Icons.event_note_rounded),
                      label: 'Riwayat',
                    ),
                    NavigationDestination(
                      icon: Icon(Icons.person_outline_rounded),
                      selectedIcon: Icon(Icons.person_rounded),
                      label: 'Profil',
                    ),
                  ],
                ),
        );
      },
    );
  }

  Future<void> _showLeaveForm() async {
    final formKey = GlobalKey<FormState>();
    final reasonController = TextEditingController();
    String leaveType = 'Cuti';
    DateTime? selectedDate;

    await showDialog<void>(
      context: context,
      builder: (dialogContext) => StatefulBuilder(
        builder: (context, setDialogState) => AlertDialog(
          title: const Text('Pengajuan izin'),
          content: SizedBox(
            width: 360,
            child: Form(
              key: formKey,
              child: Column(
                mainAxisSize: MainAxisSize.min,
                crossAxisAlignment: CrossAxisAlignment.stretch,
                children: [
                  DropdownButtonFormField<String>(
                    initialValue: leaveType,
                    decoration: const InputDecoration(labelText: 'Jenis'),
                    items: const ['Cuti', 'Izin', 'Sakit', 'WFA']
                        .map((type) => DropdownMenuItem(
                              value: type,
                              child: Text(type),
                            ))
                        .toList(),
                    onChanged: (value) =>
                        setDialogState(() => leaveType = value ?? 'Cuti'),
                  ),
                  const SizedBox(height: 12),
                  OutlinedButton.icon(
                    onPressed: () async {
                      final date = await showDatePicker(
                        context: context,
                        firstDate: DateTime.now(),
                        lastDate: DateTime.now().add(const Duration(days: 365)),
                        initialDate: DateTime.now(),
                      );
                      if (date != null) {
                        setDialogState(() => selectedDate = date);
                      }
                    },
                    icon: const Icon(Icons.calendar_today_outlined),
                    label: Text(selectedDate == null
                        ? 'Pilih tanggal'
                        : _formatDate(selectedDate!)),
                  ),
                  const SizedBox(height: 12),
                  TextFormField(
                    controller: reasonController,
                    decoration: const InputDecoration(
                      labelText: 'Alasan',
                      alignLabelWithHint: true,
                    ),
                    maxLines: 3,
                    validator: (value) => value == null || value.trim().isEmpty
                        ? 'Alasan perlu diisi'
                        : null,
                  ),
                ],
              ),
            ),
          ),
          actions: [
            TextButton(
              onPressed: () => Navigator.pop(dialogContext),
              child: const Text('Batal'),
            ),
            FilledButton(
              onPressed: () {
                if (!formKey.currentState!.validate()) return;
                Navigator.pop(dialogContext);
                ScaffoldMessenger.of(context).showSnackBar(
                  const SnackBar(content: Text('Pengajuan berhasil dikirim.')),
                );
              },
              child: const Text('Kirim pengajuan'),
            ),
          ],
        ),
      ),
    );
    reasonController.dispose();
  }
}

// Membentuk sisi kanan panel putih menjadi melengkung.
class _MobileLoginPanelClipper extends CustomClipper<Path> {
  @override
  Path getClip(Size size) {
    return Path()
      ..moveTo(0, 0)
      ..lineTo(size.width * 0.96, 0)
      ..cubicTo(
        size.width * 0.72,
        size.height * 0.13,
        size.width * 0.76,
        size.height * 0.29,
        size.width * 0.88,
        size.height * 0.43,
      )
      ..cubicTo(
        size.width * 1.02,
        size.height * 0.61,
        size.width * 0.78,
        size.height * 0.83,
        size.width * 0.96,
        size.height,
      )
      ..lineTo(0, size.height)
      ..close();
  }

  @override
  bool shouldReclip(covariant CustomClipper<Path> oldClipper) => false;
}

class _BrandMarkPainter extends CustomPainter {
  const _BrandMarkPainter();

  @override
  void paint(Canvas canvas, Size size) {
    canvas.save();
    canvas.scale(size.width / 100, size.height / 100);

    final blueRibbon = Path()
      ..moveTo(11, 37)
      ..cubicTo(21, 27, 34, 13, 43, 5)
      ..cubicTo(49, 0, 57, 1, 63, 7)
      ..lineTo(94, 37)
      ..cubicTo(99, 42, 98, 47, 93, 52)
      ..lineTo(84, 60)
      ..lineTo(60, 37)
      ..cubicTo(52, 30, 45, 28, 38, 35)
      ..lineTo(25, 48)
      ..lineTo(14, 44)
      ..cubicTo(9, 42, 7, 40, 11, 37)
      ..close();
    canvas.drawPath(
      blueRibbon,
      Paint()
        ..shader = const LinearGradient(
          begin: Alignment.topLeft,
          end: Alignment.bottomRight,
          colors: [Color(0xFF7CC3F2), Color(0xFF176BA9)],
        ).createShader(const Rect.fromLTWH(0, 0, 100, 100)),
    );

    final greenRibbon = Path()
      ..moveTo(5, 62)
      ..lineTo(17, 50)
      ..cubicTo(21, 46, 26, 46, 31, 50)
      ..lineTo(59, 76)
      ..cubicTo(65, 81, 72, 81, 78, 75)
      ..lineTo(92, 61)
      ..cubicTo(97, 56, 97, 52, 93, 48)
      ..lineTo(84, 57)
      ..lineTo(68, 73)
      ..cubicTo(64, 77, 60, 77, 55, 73)
      ..lineTo(29, 50)
      ..cubicTo(24, 46, 20, 47, 15, 52)
      ..lineTo(5, 62)
      ..cubicTo(0, 67, 1, 72, 6, 77)
      ..lineTo(28, 97)
      ..cubicTo(34, 102, 42, 101, 48, 96)
      ..lineTo(83, 62)
      ..lineTo(92, 53)
      ..lineTo(82, 61)
      ..lineTo(49, 93)
      ..cubicTo(43, 99, 37, 99, 31, 94)
      ..lineTo(8, 73)
      ..cubicTo(3, 69, 2, 66, 5, 62)
      ..close();
    canvas.drawPath(
      greenRibbon,
      Paint()
        ..shader = const LinearGradient(

String _todayLabel() {
  final now = DateTime.now();
  const weekdays = [
    'Senin',
    'Selasa',
    'Rabu',
    'Kamis',
    'Jumat',
    'Sabtu',
    'Minggu',
  ];
  const months = [
    'Januari',
    'Februari',
    'Maret',
    'April',
    'Mei',
    'Juni',
    'Juli',
    'Agustus',
    'September',
    'Oktober',
    'November',
    'Desember',
  ];
  return '${weekdays[now.weekday - 1]}, ${now.day} ${months[now.month - 1]} ${now.year}';
}

String _formatDate(DateTime date) {
  const months = [
    'Jan',
    'Feb',
    'Mar',
    'Apr',
    'Mei',
    'Jun',
    'Jul',
    'Agu',
    'Sep',
    'Okt',
    'Nov',
    'Des',
  ];
  return '${date.day} ${months[date.month - 1]} ${date.year}';
}
